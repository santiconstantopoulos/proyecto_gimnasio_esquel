import 'package:flutter_bloc/flutter_bloc.dart';

enum NavigationEvent { home, bookings, notifications, history, more }

class NavigationBloc extends Bloc<NavigationEvent, int> {
  NavigationBloc() : super(0) {
    on<NavigationEvent>((event, emit) {
      switch (event) {
        case NavigationEvent.home:
          emit(0);
          break;
        case NavigationEvent.bookings:
          emit(1);
          break;
        case NavigationEvent.notifications:
          emit(2);
          break;
        case NavigationEvent.history:
          emit(3);
          break;
        case NavigationEvent.more:
          emit(4);
          break;
      }
    });
  }
}
